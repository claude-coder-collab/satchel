cmake_minimum_required(VERSION 3.28)

project(satchel VERSION 0.1.0 DESCRIPTION "Lossless media packaging as standard zip archives" LANGUAGES C CXX)

if(MSVC)
    set(ZP_CXX_STANDARD 23)
else()
    set(ZP_CXX_STANDARD 26)
endif()
set(CMAKE_CXX_STANDARD ${ZP_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(ZP_VERSION "" CACHE STRING "Version string embedded in binaries and archives (empty: the project version)")
if(NOT ZP_VERSION)
    set(ZP_VERSION "${PROJECT_VERSION}")
endif()

include(cmake/ProjectOptions.cmake)
include(cmake/Dependencies.cmake)
include(cmake/Packaging.cmake)

if(ZP_BUILD_TESTS)
    enable_testing()
endif()

add_subdirectory(core)
if(EMSCRIPTEN)
    set(ZP_WASM_MAXIMUM_MEMORY "512MB" CACHE STRING "WebAssembly memory limit of the browser module")
    add_subdirectory(bindings/js)
endif()

option(ZP_BUILD_CLI "Build the command-line tool" ON)
if(ZP_BUILD_CLI AND NOT EMSCRIPTEN)
    add_subdirectory(desktop/cli)
endif()

if(ZP_BUILD_TESTS)
    add_subdirectory(tests)
endif()

if(NOT EMSCRIPTEN)
    add_subdirectory(desktop/gui)
    if(WIN32)
        add_subdirectory(desktop/windows)
    endif()
    option(ZP_BUILD_QUICKLOOK "Build the macOS Quick Look extension (needs Swift)" ON)
    if(APPLE AND TARGET satchel_gui AND ZP_BUILD_QUICKLOOK)
        add_subdirectory(desktop/macos)
    endif()
    if(TARGET satchel_cli)
        if(APPLE AND TARGET satchel_gui)
            install(TARGETS satchel_cli RUNTIME DESTINATION Satchel.app/Contents/Helpers)
        else()
            install(TARGETS satchel_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
        endif()
    endif()
    zp_configure_cpack()
endif()

if(ZP_BUILD_FUZZERS)
    add_subdirectory(tests/fuzz)
endif()

option(ZP_BUILD_BENCH "Build the throughput benchmark (zp_bench)" ON)
if(ZP_BUILD_BENCH AND NOT ZP_BUILD_FUZZERS)
    add_subdirectory(bench)
endif()
